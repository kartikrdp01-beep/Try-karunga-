-- Additional hue presets are generated as distinct selectable palettes. The
-- selection list stays compact in code while still providing a large catalogue.
local function HsvColor(hue, saturation, value)
    return Color3.fromHSV((hue % 360) / 360, saturation, value)
end

for index = 1, 48 do
    local hue = (index * 137.508) % 360
    local accent = HsvColor(hue, 0.78, 1)
    local accent2 = HsvColor(hue + 24, 0.48, 1)
    local bg = HsvColor(hue + 196, 0.32, 0.075)
    local panel = HsvColor(hue + 196, 0.30, 0.13)
    local surface = HsvColor(hue + 196, 0.28, 0.19)
    table.insert(Themes, {
        Name = string.format("Spectrum %02d", index),
        Bg = bg,
        Panel = panel,
        Surface = surface,
        Accent = accent,
        Accent2 = accent2,
        Text = Color3.fromRGB(249, 248, 255),
        Muted = Color3.fromRGB(166, 163, 185),
    })
end

--// Utility functions ----------------------------------------------------------
local function Clamp(value, low, high)
    return math.max(low, math.min(high, value))
end

local function SafeDestroy(instance)
    if instance and instance.Parent then
        pcall(function()
            instance:Destroy()
        end)
    end
end

local function Create(className, props, parent)
    local object = Instance.new(className)
    for property, value in pairs(props or {}) do
        local ok, err = pcall(function()
            object[property] = value
        end)
        if not ok then
            warn("[EMILY RECHAT] Property skipped:", property, err)
        end
    end
    if parent then
        object.Parent = parent
    end
    return object
end

local function Round(parent, radius)
    return Create("UICorner", {CornerRadius = UDim.new(0, radius or 10)}, parent)
end

local function Outline(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or State.Theme.Accent,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function AddPadding(parent, left, top, right, bottom)
    return Create("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
    }, parent)
end

local function AddList(parent, direction, padding, horizontalAlignment, verticalAlignment)
    return Create("UIListLayout", {
        FillDirection = direction or Enum.FillDirection.Vertical,
        SortOrder = Enum.SortOrder.LayoutOrder,
        Padding = UDim.new(0, padding or 0),
        HorizontalAlignment = horizontalAlignment or Enum.HorizontalAlignment.Left,
        VerticalAlignment = verticalAlignment or Enum.VerticalAlignment.Top,
    }, parent)
end

local function AddGrid(parent, cellSize, cellPadding, fillDirectionMaxCells)
    return Create("UIGridLayout", {
        CellSize = cellSize,
        CellPadding = cellPadding or UDim2.fromOffset(8, 8),
        FillDirectionMaxCells = fillDirectionMaxCells or 4,
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, parent)
end

local function Fade(instance, properties, duration, style, direction)
    local tween = TweenService:Create(instance, TweenInfo.new(
        duration or 0.2,
        style or Enum.EasingStyle.Quad,
        direction or Enum.EasingDirection.Out
    ), properties)
    tween:Play()
    return tween
end

local function MakeButton(parent, text, size, callback, options)
    options = options or {}
    local button = Create("TextButton", {
        AutoButtonColor = false,
        BackgroundColor3 = options.Background or State.Theme.Surface,
        BackgroundTransparency = options.Transparency or 0,
        BorderSizePixel = 0,
        Size = size or UDim2.fromOffset(100, 34),
        Font = options.Font or Enum.Font.GothamSemibold,
        Text = text or "Button",
        TextColor3 = options.TextColor or State.Theme.Text,
        TextSize = options.TextSize or 12,
        TextWrapped = true,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, parent)
    Round(button, options.Radius or 9)
    if options.Stroke then
        Outline(button,
            options.StrokeColor or State.Theme.Accent,
            options.StrokeThickness or 1,
            options.StrokeTransparency or 0.2)
    end
    button.MouseEnter:Connect(function()
        if button.Parent then
            Fade(button, {BackgroundColor3 = options.Hover or State.Theme.Accent}, 0.12)
        end
    end)
    button.MouseLeave:Connect(function()
        if button.Parent then
            Fade(button, {BackgroundColor3 = options.Background or State.Theme.Surface}, 0.12)
        end
    end)
    button.Activated:Connect(function()
        if State.SoundEnabled then
            -- No custom sound asset is required; the interface stays silent unless
            -- the user later chooses a sound in Roblox's native settings.
        end
        if callback then
            local ok, err = pcall(callback)
            if not ok then
                warn("[EMILY RECHAT] Button action failed:", err)
            end
        end
    end)
    return button
end

local function FormatClock(timestamp)
    local date = os.date("!*t", timestamp or os.time())
    local offset = os.difftime(os.time(), os.time(date))
    local localDate = os.date("*t", timestamp or os.time())
    return string.format("%02d:%02d", localDate.hour, localDate.min)
end

local function Truncate(text, limit)
    text = tostring(text or "")
    if #text > limit then
        return string.sub(text, 1, limit - 1) .. "…"
    end
    return text
end

local function NormalizeKey(text)
    return string.lower(tostring(text or "")):gsub("%s+", "")
end

local function IsTouchOnly()
    return UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
end

local function GetPlayerByName(fragment)
    fragment = string.lower(tostring(fragment or ""))
    if fragment == "" then
        return nil
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if string.lower(player.Name) == fragment or string.lower(player.DisplayName) == fragment then
            return player
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if string.find(string.lower(player.Name), fragment, 1, true)
            or string.find(string.lower(player.DisplayName), fragment, 1, true) then
            return player
        end
    end
    return nil
end

--// Notifications --------------------------------------------------------------
local function Notify(title, message, kind, duration)
    if not State.ShowNotifications then
        return
    end
    NotificationSerial += 1
    local serial = NotificationSerial
    local accent = State.Theme.Accent
    if kind == "success" then
        accent = Color3.fromRGB(74, 219, 139)
    elseif kind == "warning" then
        accent = Color3.fromRGB(255, 184, 75)
    elseif kind == "error" then
        accent = Color3.fromRGB(239, 81, 105)
    end
    local toast = Create("Frame", {
        Name = "Toast_" .. tostring(serial),
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 380, 0, 95 + ((serial % 4) * 82)),
        Size = UDim2.fromOffset(300, 72),
        BackgroundColor3 = State.Theme.Panel,
        BackgroundTransparency = 0.02,
        BorderSizePixel = 0,
        ZIndex = 102,
    }, ToastLayer)
    Round(toast, 12)
    Outline(toast, accent, 1.2, 0.08)
    local bar = Create("Frame", {
        BackgroundColor3 = accent,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(0, 12),
        Size = UDim2.new(0, 3, 1, -24),
        ZIndex = 103,
    }, toast)
    Round(bar, 2)
    MakeLabel(toast,
        Truncate(title,
        42),
        UDim2.new(1,
        -25,
        0,
        23),
        UDim2.fromOffset(14,
        8),
        12,
        State.Theme.Text,
        Enum.Font.GothamBold).ZIndex = 103
    MakeLabel(toast,
        Truncate(message,
        115),
        UDim2.new(1,
        -25,
        0,
        30),
        UDim2.fromOffset(14,
        33),
        10,
        State.Theme.Muted).ZIndex = 103
    Fade(toast,
        {Position = UDim2.new(1,
        -16,
        0,
        95 + ((serial % 4) * 82))},
        0.24,
        Enum.EasingStyle.Back)
    task.delay(duration or 3.2, function()
        if toast.Parent then
            Fade(toast,
                {Position = UDim2.new(1,
                340,
                0,
                toast.Position.Y.Offset),
                BackgroundTransparency = 1},
                0.18)
            task.delay(0.22, function()
                SafeDestroy(toast)
            end)
        end
    end)
end

--// Shared scroll-frame helper -------------------------------------------------
local function MakeScroll(parent, name, position, size, options)
    options = options or {}
    local scroll = Create("ScrollingFrame", {
        Name = name,
        BackgroundColor3 = options.Background or State.Theme.Bg,
        BackgroundTransparency = options.Transparency == nil and 0.12 or options.Transparency,
        BorderSizePixel = 0,
        Position = position or UDim2.new(),
        Size = size or UDim2.fromScale(1, 1),
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollingDirection = Enum.ScrollingDirection.Y,
        ScrollBarThickness = options.ScrollBarThickness or 4,
        ScrollBarImageColor3 = State.Theme.Accent,
        ElasticBehavior = Enum.ElasticBehavior.WhenScrollable,
    }, parent)
    Round(scroll, options.Radius or 10)
    AddPadding(scroll,
        options.PaddingLeft or 8,
        options.PaddingTop or 8,
        options.PaddingRight or 8,
        options.PaddingBottom or 8)
    local layout = AddList(scroll,
        options.Direction or Enum.FillDirection.Vertical,
        options.Padding or 7)
    layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        scroll.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 18)
    end)
    return scroll, layout
end

local function SaveMessage(data)
    table.insert(State.MessageHistory, data)
    if #State.MessageHistory > MAX_STORED_MESSAGES then
        table.remove(State.MessageHistory, 1)
    end
end

local function MatchesMessageSearch(data)
    local query = string.lower(State.SearchText or "")
    if query == "" then
        return true
    end
    return string.find(string.lower(data.Text or ""), query, 1, true) ~= nil
        or string.find(string.lower(data.Username or ""), query, 1, true) ~= nil
        or string.find(string.lower(data.DisplayName or ""), query, 1, true) ~= nil
end

local function ScrollChatToBottom()
    if not ChatScroll then
        return
    end
    task.defer(function()
        if ChatScroll and ChatScroll.Parent then
            ChatScroll.CanvasPosition = Vector2.new(0,
                math.max(0,
                ChatScroll.AbsoluteCanvasSize.Y - ChatScroll.AbsoluteSize.Y))
        end
    end)
end

local function RebuildChatView()
    if not ChatScroll then
        return
    end
    for _, child in ipairs(ChatScroll:GetChildren()) do
        if child:IsA("Frame") or child:IsA("TextLabel") then
            child:Destroy()
        end
    end
    State.MessageObjects = {}
    local shown = 0
    for _, data in ipairs(State.MessageHistory) do
        if MatchesMessageSearch(data) then
            shown += 1
            if shown > MAX_VISIBLE_MESSAGES then
                break
            end
            if LocalMessageRenderer then
                LocalMessageRenderer(data, false)
            end
        end
    end
    if MessageEmptyState then
        MessageEmptyState.Visible = shown == 0
    end
    ScrollChatToBottom()
end

local function AddMessage(username, displayName, text, kind, extra)
    extra = extra or {}
    local record = {
        Id = State.MessageCount + 1,
        Username = username or "System",
        DisplayName = displayName or username or "System",
        Text = tostring(text or ""),
        Kind = kind or "chat",
        Timestamp = os.time(),
        TimeText = FormatClock(os.time()),
        Channel = extra.Channel or "RBXGeneral",
        UserId = extra.UserId,
        IsLocal = username == LocalPlayer.Name,
    }
    State.MessageCount += 1
    record.Id = State.MessageCount
    SaveMessage(record)
    if #State.MessageHistory <= MAX_VISIBLE_MESSAGES and ChatScroll and (State.SearchText == "" or MatchesMessageSearch(record)) then
        LocalMessageRenderer(record)
        MessageEmptyState.Visible = false
        ScrollChatToBottom()
    else
        RebuildChatView()
    end
    if ChatMessageCounter then
        ChatMessageCounter.Text = tostring(State.MessageCount) .. " messages"
    end
    if not IsLocalAuthor(record.Username) and kind ~= "system" and State.ActiveTab ~= "CHAT" and State.ActiveTab ~= "DM" then
        State.Unread += 1
        if CurrentTabBadge then
            CurrentTabBadge.Text = tostring(State.Unread)
            CurrentTabBadge.Visible = State.Unread > 0
        end
    end
    return record
end

--// Composer / sending ---------------------------------------------------------
local function GetGeneralChannel()
    local channels = TextChatService:FindFirstChild("TextChannels")
    if not channels then
        return nil
    end
    return channels:FindFirstChild("RBXGeneral") or channels:FindFirstChildWhichIsA("TextChannel")
end

local function IsTextChatReady()
    return TextChatService.ChatVersion == Enum.ChatVersion.TextChatService
end

local function SendToRobloxChat(text, targetName)
    text = tostring(text or "")
    if text:gsub("%s", "") == "" then
        return false, "Write a message first."
    end
    if #text > MAX_MESSAGE_LENGTH then
        return false,
            "Message is too long. Limit: " .. tostring(MAX_MESSAGE_LENGTH) .. " characters."
    end
    local now = os.clock()
    if State.SendPending then
        return false, "A message is already being sent."
    end
    local remaining = SEND_COOLDOWN - (now - State.LastSendAt)
    if remaining > 0 then
        return false, string.format("Please wait %.1fs before sending again.", remaining)
    end
    if not IsTextChatReady() then
        return false, "This experience is using legacy chat. Use Roblox's native chat controls."
    end
    local channel = GetGeneralChannel()
    if not channel then
        return false, "The default chat channel is not ready yet."
    end
    local outgoing = text
    if targetName and targetName ~= "" then
        -- This uses TextChatService's native whisper command path. The experience
        -- still applies its own chat permissions and filtering rules.
        outgoing = "/w " .. targetName .. " " .. text
    end
    State.SendPending = true
    State.LastSendAt = now
    local ok, err = pcall(function()
        channel:SendAsync(outgoing)
    end)
    task.delay(SEND_COOLDOWN, function()
        State.SendPending = false
    end)
    if not ok then
        State.SendPending = false
        return false, "Roblox chat did not accept the message: " .. tostring(err)
    end
    return true
end

local function SendComposerMessage(textOverride)
    local text = textOverride
    if text == nil and Composer then
        text = Composer.Text
    end
    text = tostring(text or "")
    if text:gsub("%s", "") == "" then
        return
    end

    local first, remainder = string.match(text, "^/(%S+)%s*(.*)$")
    if first then
        local command = string.lower(first)
        if command == "clear" then
            State.MessageHistory = {}
            State.MessageObjects = {}
            State.MessageCount = 0
            RebuildChatView()
            AddSystemMessage("Chat panel cleared locally.")
            if Composer then Composer.Text = "" end
            return
        elseif command == "help" then
            Notify("Command help",
                "/ai, /clear, /nick, /bio, /theme, /mute, /unmute, /reply",
                "info",
                5)
            if Composer then Composer.Text = "" end
            return
        elseif command == "ai" then
            if Composer then Composer.Text = "" end
            if UpdateNavigation then
                UpdateNavigation("AI")
            end
            return
        elseif command == "nick" then
            if remainder and remainder ~= "" then
                State.RPName = Truncate(remainder, 30)
                if ProfileNameInput then ProfileNameInput.Text = State.RPName end
                RebuildChatView()
                AddSystemMessage("Local profile name updated. This does not change your Roblox username.")
            else
                Notify("Profile name", "Usage: /nick Your RP Name", "warning")
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "bio" then
            if remainder and remainder ~= "" then
                State.Bio = Truncate(remainder, 90)
                if ProfileBioInput then ProfileBioInput.Text = State.Bio end
                AddSystemMessage("Local profile bio updated.")
            else
                Notify("Profile bio", "Usage: /bio your bio", "warning")
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "theme" then
            if remainder and remainder ~= "" then
                local search = string.lower(remainder)
                local found = nil
                for i, theme in ipairs(Themes) do
                    if string.lower(theme.Name) == search or string.find(string.lower(theme.Name),
                        search,
                        1,
                        true) then
                        found = i
                        break
                    end
                end
                if found then
                    State.ThemeIndex = found
                    State.Theme = Themes[found]
                    -- ApplyTheme is defined after the theme controller is built.
                    if _G.__EMILY_RECHAT_APPLY_THEME then
                        _G.__EMILY_RECHAT_APPLY_THEME(found)
                    end
                    AddSystemMessage("Theme changed to " .. Themes[found].Name .. ".")
                else
                    Notify("Theme not found",
                        "Open THEME to browse the available palettes.",
                        "warning")
                end
            else
                if UpdateNavigation then
                    UpdateNavigation("THEME")
                end
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "mute" then
            local target = GetPlayerByName(remainder)
            if target then
                State.IgnoredUsers[target.Name] = true
                AddSystemMessage("Messages from @" .. target.Name .. " will be hidden in this panel.")
            else
                Notify("Mute user", "Usage: /mute username", "warning")
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "unmute" then
            local target = GetPlayerByName(remainder)
            local key = target and target.Name or remainder
            if key and key ~= "" then
                State.IgnoredUsers[key] = nil
                AddSystemMessage("Local mute removed for " .. key .. ".")
            else
                Notify("Unmute user", "Usage: /unmute username", "warning")
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "reply" then
            local target = GetPlayerByName(remainder)
            if target and Composer then
                Composer.Text = "@" .. target.Name .. " "
                Composer:CaptureFocus()
                Composer.CursorPosition = #Composer.Text + 1
            else
                Notify("Reply", "Usage: /reply username", "warning")
            end
            return
        elseif command == "sticker" then
            if UpdateNavigation then
                UpdateNavigation("GALLERY")
            end
            if Composer then Composer.Text = "" end
            return
        elseif command == "me" then
            if remainder and remainder ~= "" then
                AddMessage(LocalPlayer.Name,
                    State.RPName,
                    "* " .. State.RPName .. " " .. remainder,
                    "action")
                if Composer then Composer.Text = "" end
            else
                Notify("Action", "Usage: /me waves", "warning")
            end
            return
        end
    end

    local ok, errorMessage = SendToRobloxChat(text)
    if ok then
        if Composer then
            Composer.Text = ""
            Composer:ReleaseFocus()
        end
        Notify("Message queued", "Sent through the Roblox chat service.", "success", 1.8)
    else
        Notify("Couldn't send", errorMessage, "warning", 3.2)
    end
end

local function RebuildDMPlayers(filterText)
    for _, child in ipairs(DMPlayerScroll:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    local filter = string.lower(filterText or "")
    local count = 0
    for _, target in ipairs(Players:GetPlayers()) do
        if target ~= LocalPlayer then
            local nameLine = string.lower(target.Name .. " " .. target.DisplayName)
            if filter == "" or string.find(nameLine, filter, 1, true) then
                count += 1
                local row = Create("Frame", {
                    BackgroundColor3 = State.SelectedDM == target and State.Theme.Surface or State.Theme.Panel,
                    BorderSizePixel = 0,
                    Size = UDim2.new(1, -2, 0, 40),

                    BackgroundColor3 = State.Theme.Surface,
                    BorderSizePixel = 0,
                    Position = UDim2.fromOffset(5, 4),
                    Size = UDim2.fromOffset(32, 32),
                    Image = "",
                }, row)
                Round(avatar, 16)
                task.spawn(function()
                    local ok, image = pcall(function()
                        return Players:GetUserThumbnailAsync(target.UserId,
                            Enum.ThumbnailType.HeadShot,
                            Enum.ThumbnailSize.Size48x48)
                    end)
                    if ok and avatar.Parent then avatar.Image = image end
                end)
                MakeLabel(row,
                    target.DisplayName,
                    UDim2.new(1,
                    -125,
                    0,
                    18),
                    UDim2.fromOffset(44,
                    4),
                    10,
                    State.Theme.Text,
                    Enum.Font.GothamBold)
                MakeLabel(row,
                    "@" .. target.Name,
                    UDim2.new(1,
                    -125,
                    0,
                    14),
                    UDim2.fromOffset(44,
                    21),
                    9,

local function AIResponse(prompt)
    local text = string.lower(tostring(prompt or ""))
    if string.find(text,
        "hello",
        1,
        true) or string.find(text,
        "greeting",
        1,
        true) or string.find(text,
        "hi ",
        1,
        true) then
        return AIFallbacks.Greeting[math.random(1, #AIFallbacks.Greeting)]
    elseif string.find(text,
        "polite",
        1,
        true) or string.find(text,
        "rewrite",
        1,
        true) or string.find(text,
        "formal",
        1,
        true) then
        return AIFallbacks.Polite[math.random(1, #AIFallbacks.Polite)]
    elseif string.find(text,
        "roleplay",
        1,
        true) or string.find(text,
        "scene",
        1,
        true) or string.find(text,
        "rp ",
        1,
        true) then
        return AIFallbacks.Roleplay[math.random(1, #AIFallbacks.Roleplay)]
    elseif string.find(text, "starter", 1, true) or string.find(text, "conversation", 1, true) then
        return AIFallbacks.Starter[math.random(1, #AIFallbacks.Starter)]
    elseif string.find(text, "emoji", 1, true) or string.find(text, "sticker", 1, true) then
        return AIFallbacks.Emoji[math.random(1, #AIFallbacks.Emoji)]
    elseif string.find(text, "command", 1, true) or string.find(text, "/help", 1, true) then
        return "Commands: /help, /clear, /ai, /nick NAME, /bio TEXT, /theme NAME, /mute USER, /unmute USER, /reply USER, /sticker, /me ACTION."
    elseif string.find(text, "your name", 1, true) or string.find(text, "who are you", 1, true) then
        return "I'm FC AI, the local helper inside EMILY RECHAT. I work offline and don't send prompts to an API."
    end
    return AIFallbacks.General[math.random(1, #AIFallbacks.General)]
end

-- Theme application is centralized so all live controls can be updated together.
local function ApplyTheme(index)
    index = Clamp(tonumber(index) or 1, 1, #Themes)
    State.ThemeIndex = index
    State.Theme = Themes[index]
    local theme = State.Theme
    local propertyColors = {
        [MainWindow] = {BackgroundColor3 = theme.Bg},
        [TopBar] = {BackgroundColor3 = theme.Panel},
        [TopBarMask] = {BackgroundColor3 = theme.Panel},
        [Logo] = {BackgroundColor3 = theme.Surface},
        [StatusPill] = {BackgroundColor3 = theme.Surface},
        [Sidebar] = {BackgroundColor3 = theme.Panel},
        [SidebarFooter] = {BackgroundColor3 = theme.Surface},
        [ChatPanel] = {BackgroundColor3 = theme.Bg},
        [ComposerBar] = {BackgroundColor3 = theme.Panel},
        [ChannelTag] = {BackgroundColor3 = theme.Surface},
        [ChatSearch] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [Composer] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [DMComposer] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [DMSearch] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [AIComposer] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [ThemeSearch] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
        [GallerySearch] = {BackgroundColor3 = theme.Surface,
            PlaceholderColor3 = theme.Muted,
            TextColor3 = theme.Text},
    }
    for object, properties in pairs(propertyColors) do
        if object and object.Parent then
            for property, value in pairs(properties) do
                pcall(function() object[property] = value end)
            end
        end
    end
    pcall(function() MainStroke.Color = theme.Accent end)
    pcall(function() LogoStroke.Color = theme.Accent end)
    pcall(function() MainGradient.Color = ColorSequence.new(theme.Accent, theme.Accent2) end)
    local colorTargets = {
        {Title, theme.Text}, {Subtitle, theme.Muted}, {StatusText, theme.Text},
        {SidebarTitle, theme.Muted}, {FooterCreator, theme.Muted}, {FooterName, theme.Accent2},
        {PageTitle, theme.Text}, {PageDescription, theme.Muted}, {ChatHeaderStatus, theme.Muted},
        {ChatMessageCounter,
            theme.Muted},
            {ChatSafetyText,
            theme.Muted},
            {DMTargetLabel,
            theme.Muted},
        {DMHint, theme.Muted}, {MessageEmptyState, theme.Muted},
    }
    for _, pair in ipairs(colorTargets) do
        local object, color = pair[1], pair[2]
        if object and object.Parent then
            pcall(function() object.TextColor3 = color end)
        end
    end
    for _, button in pairs(NavButtons) do
        if button and button.Parent then
            button.BackgroundColor3 = State.ActiveTab == button.Name and theme.Surface or theme.Panel
            button.TextColor3 = State.ActiveTab == button.Name and theme.Accent2 or theme.Muted
        end
    end
    for _, child in ipairs(ToastLayer:GetChildren()) do
        if child:IsA("Frame") then
            child.BackgroundColor3 = theme.Panel
        end
    end
    RebuildChatView()
    if ThemeGridFrame then
        -- Rebuild the palette preview to ensure the active border is accurate.
        local search = ThemeSearch.Text
        task.defer(function()
            if ThemeSearch.Parent then
                ThemeSearch.Text = search
            end
        end)
    end
end

local function BuildGallery(filterText)
    ClearContainer(GalleryScroll, true)
    local filter = string.lower(filterText or "")
    for index, sticker in ipairs(Stickers) do
        if filter == "" or string.find(string.lower(sticker.Name .. " " .. sticker.Emoji),
            filter,
            1,
            true) then
            local card = Create("TextButton", {
                AutoButtonColor = false,
                BackgroundColor3 = State.Theme.Panel,
                BorderSizePixel = 0,
                Size = UDim2.new(0.25, -7, 0, 76),
                Text = "",
                LayoutOrder = index,
            }, GalleryScroll)
            Round(card, 10)
            Outline(card, State.Theme.Surface, 1, 0.2)
            MakeLabel(card,
                sticker.Emoji,
                UDim2.new(1,
                0,
                0,
                43),
                UDim2.new(),
                23,
                State.Theme.Text,
                Enum.Font.Gotham)
            local itemName = MakeLabel(card,
                sticker.Name,
                UDim2.new(1,
                -8,
                0,
                18),
                UDim2.new(0,
                4,
                1,
                -21),
                8,
                State.Theme.Muted,
                Enum.Font.GothamSemibold)
            itemName.TextXAlignment = Enum.TextXAlignment.Center
            card.Activated:Connect(function()
                if Composer then
                    Composer.Text = (Composer.Text or "") .. sticker.Emoji
                    Composer:CaptureFocus()
                    Composer.CursorPosition = #Composer.Text + 1
                end
                Notify("Sticker inserted",
                    sticker.Name .. " added to your composer.",
                    "success",
                    1.5)
            end)
        end
    end
end
GallerySearch:GetPropertyChangedSignal("Text"):Connect(function()
    BuildGallery(GallerySearch.Text)
end)

        end

        State.Unread = 0
        CurrentTabBadge.Visible = false
    end
end

--// Drag handling --------------------------------------------------------------
local function BeginDrag(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        Dragging = true
        DragStart = input.Position
        StartPosition = MainWindow.Position
        CurrentDragInput = input
    end
end

local function UpdateDrag(input)
    if Dragging and StartPosition and DragStart then
        local delta = input.Position - DragStart
        local newPosition = UDim2.new(
            StartPosition.X.Scale,
            StartPosition.X.Offset + delta.X,
            StartPosition.Y.Scale,
            StartPosition.Y.Offset + delta.Y
        )
        MainWindow.Position = newPosition
    end
end

local function EndDrag(input)
    if input == CurrentDragInput or input.UserInputType == Enum.UserInputType.MouseButton1 then
        Dragging = false
        CurrentDragInput = nil
        DragStart = nil
        StartPosition = nil
    end
end

TopBar.InputBegan:Connect(BeginDrag)
UserInputService.InputChanged:Connect(function(input)
    if Dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        UpdateDrag(input)
    end
end)
UserInputService.InputEnded:Connect(EndDrag)

    FloatingButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            floatingDragging = true
            floatingStart = input.Position
            floatingPos = FloatingButton.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if floatingDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - floatingStart

                floatingPos.Y.Scale,

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

--// Responsive layout ----------------------------------------------------------
function UpdateResponsiveScale()
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end
    local viewport = camera.ViewportSize
    local targetScale = math.min(viewport.X / (WINDOW_DESIGN_SIZE.X + 36),
        viewport.Y / (WINDOW_DESIGN_SIZE.Y + 60))
    targetScale = Clamp(targetScale, 0.34, 1)
    if IsTouchOnly() then
        targetScale = math.min(targetScale, 0.92)
        Sidebar.Size = UDim2.new(0, 129, 1, -94)
        Content.Position = UDim2.fromOffset(151, 82)
        Content.Size = UDim2.new(1, -163, 1, -94)
    else
        Sidebar.Size = UDim2.new(0, 155, 1, -94)
        Content.Position = UDim2.fromOffset(178, 82)
        Content.Size = UDim2.new(1, -190, 1, -94)
    end
    -- Keep the window centred only when it is already near the centre; user drag
    -- coordinates otherwise remain intact.
    if math.abs(MainWindow.Position.X.Scale - 0.5) < 0.03 and math.abs(MainWindow.Position.Y.Scale - 0.5) < 0.03 then
        MainWindow.Position = UDim2.fromScale(0.5, 0.5)
    end
    RootScale.Scale = Clamp(targetScale * UserScale, 0.34, 1.30)
end

if workspace.CurrentCamera then
    workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(UpdateResponsiveScale)
end
workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    task.wait(0.15)
    if workspace.CurrentCamera then
        workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(UpdateResponsiveScale)
    end
    UpdateResponsiveScale()
end)

--// Incoming chat integration --------------------------------------------------
local function OnTextChatMessage(message)
    if not message then return end
    local text = tostring(message.Text or "")
    if text == "" then return end
    local source = message.TextSource
    if not source then
        AddMessage("System",
            "Roblox",
            text,
            "system",
            {Channel = message.TextChannel and message.TextChannel.Name})
        return
    end
    local sender = Players:GetPlayerByUserId(source.UserId)
    local username = sender and sender.Name or ("User" .. tostring(source.UserId))
    if State.IgnoredUsers[username] then
        return
    end
    local displayName = sender and sender.DisplayName or username
    AddMessage(username, displayName, text, "chat", {
        UserId = source.UserId,
        Channel = message.TextChannel and message.TextChannel.Name,
    })
end

local messageReceivedConnected = false
pcall(function()
    TextChatService.MessageReceived:Connect(OnTextChatMessage)
    messageReceivedConnected = true
end)
if not messageReceivedConnected then
    warn("[EMILY RECHAT] TextChatService.MessageReceived is not available in this experience.")
end

            Composer:ReleaseFocus()

            MinimizeWindow()
        end
    end
end))
